    <h2>🎬 រឿងភាគចិនខ្លីៗ</h2>
    
    <div class="user-info" id="userInfo">
        កំពុងផ្ទុកទិន្នន័យ User...
    </div>

    <div id="movieList">
        <p style="text-align: center;">កំពុងទាញយករឿង...</p>
    </div>

    <script>
        // បើក Telegram WebApp SDK
        let tg = window.Telegram.WebApp;
        tg.expand();

        // យក Telegram User ID របស់ភ្ញៀវ
        let user = tg.initDataUnsafe.user;
        let telegramId = user ? user.id : "GUEST_123"; // ID ត្រឹមត្រូវពេលរត់ក្នុង Telegram
        let username = user ? user.username : "Guest";

        document.getElementById('userInfo').innerHTML = `👤 <b>ស្វាគមន៍៖</b> ${user ? user.first_name : 'ភ្ញៀវ'} | ID: ${telegramId}`;

        // ដាក់ Google Apps Script Web App URL ของคุณទីនេះ
        const API_URL = "https://script.google.com/macros/s/AKfycbw84JSDD3mYSD2XwrqXLqOASDdIDfwhvIgbv2p4Zy5pH7w1ChCg47VXS9_prjEgMUcI/exec";

        let userStatus = "Free";

        // ពិនិត្យមើលសមាជិកភាព User
        function checkUserStatus() {
            fetch(`${API_URL}?action=checkUser&telegramId=${telegramId}`)
                .then(response => response.json())
                .then(data => {
                    if(data.status === "success") {
                        userStatus = data.data.status; // Free ឬ VIP
                        if(userStatus === "VIP") {
                            document.getElementById('userInfo').innerHTML += `<br>⭐ ស្ថានភាព៖ <span style="color: #eab308; font-weight: bold;">VIP Member</span> (ផុតកំណត់៖ ${data.data.expiryDate})`;
                        } else {
                            document.getElementById('userInfo').innerHTML += `<br>⭐ ស្ថានភាព៖ Free (សូមអភិវឌ្ឍជា VIP ដើម្បីមើលរឿងពេញ)`;
                        }
                    }
                    loadMovies();
                })
                .catch(err => {
                    console.error(err);
                    loadMovies();
                });
        }

        // ទាញយកបញ្ជីរឿង
        function loadMovies() {
            fetch(`${API_URL}?action=getMovies`)
                .then(response => response.json())
                .then(data => {
                    if(data.status === "success") {
                        let movies = data.data;
                        let html = "";
                        
                        movies.forEach(movie => {
                            let isVipMovie = movie.type === "VIP";
                            let canWatch = !isVipMovie || (userStatus === "VIP");
                            
                            html += `
                                <div class="movie-card">
                                    <img src="${movie.thumbnail}" alt="${movie.title}">
                                    <div class="movie-content">
                                        <div class="movie-title">${movie.title} - ភាគទី ${movie.episode}</div>
                                        <span class="badge ${isVipMovie ? 'badge-vip' : 'badge-free'}">${movie.type}</span>
                                        ${canWatch ? 
                                            `<a href="${movie.videoUrl}" target="_blank" class="btn">▶️ មើលវីដេអូ</a>` : 
                                            `<button class="btn btn-vip" onclick="alert('រឿងនេះសម្រាប់តែ VIP! សូមទាក់ទងបង់ប្រាក់ដើម្បីទស្សនា')">🔒 រឿង VIP (ចង់ទិញ)</button>`
                                        }
                                    </div>
                                </div>
                            `;
                        });
                        document.getElementById('movieList').innerHTML = html;
                    }
                })
                .catch(err => {
                    document.getElementById('movieList').innerHTML = "<p style='text-align: center; color: red;'>បរាជ័យក្នុងការទាញទិន្នន័យរឿង</p>";
                });
        }

        // ចាប់ផ្តើមដំណើរការ
        checkUserStatus();
    </script>
